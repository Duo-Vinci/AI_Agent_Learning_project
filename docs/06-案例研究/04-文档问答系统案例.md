## 三、核心功能

### 3.1 文档解析

```python
from langchain_community.document_loaders import PyPDFLoader, Docx2txtLoader

class DocumentParser:
    def parse(self, file_path: str, file_type: str):
        if file_type == 'pdf':
            loader = PyPDFLoader(file_path)
        elif file_type == 'docx':
            loader = Docx2txtLoader(file_path)
        
        return loader.load()
```

### 3.2 向量索引

```python
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import HuggingFaceEmbeddings

class VectorIndexBuilder:
    def __init__(self):
        self.embeddings = HuggingFaceEmbeddings(
            model_name="BAAI/bge-large-zh-v1.5"
        )
    
    def build(self, documents):
        return FAISS.from_documents(documents, self.embeddings)
```

### 3.3 问答引擎

```python
class DocumentQAEngine:
    def __init__(self, vectorstore):
        self.llm = ChatOpenAI(model="deepseek-chat")
        self.vectorstore = vectorstore
    
    async def answer(self, question: str):
        docs = self.vectorstore.similarity_search(question, k=4)
        
        context = "\n\n".join([doc.page_content for doc in docs])
        
        prompt = f"""根据以下文档回答问题：

{context}

问题：{question}
要求：基于文档内容，标注来源

回答："""
        
        answer = await self.llm.agenerate(prompt)
        
        return {
            'answer': answer,
            'sources': [doc.metadata['source'] for doc in docs]
        }
```

### 3.4 完整系统

```python
class DocumentQASystem:
    def __init__(self):
        self.parser = DocumentParser()
        self.index_builder = VectorIndexBuilder()
        self.qa_engine = None
    
    async def ingest_document(self, file_path: str, file_type: str):
        # 解析文档
        documents = self.parser.parse(file_path, file_type)
        
        # 构建索引
        vectorstore = self.index_builder.build(documents)
        
        # 初始化QA引擎
        self.qa_engine = DocumentQAEngine(vectorstore)
        
        return {'status': 'success', 'docs': len(documents)}
    
    async def query(self, question: str):
        return await self.qa_engine.answer(question)
```

## 四、权限控制

```python
class PermissionManager:
    def __init__(self):
        self.permissions = {}
    
    def grant(self, user_id: str, doc_ids: list):
        self.permissions[user_id] = doc_ids
    
    def check(self, user_id: str, doc_id: str) -> bool:
        return doc_id in self.permissions.get(user_id, [])
    
    def filter_results(self, user_id: str, docs):
        return [
            doc for doc in docs
            if self.check(user_id, doc.metadata['doc_id'])
        ]
```

## 五、API服务

```python
from fastapi import FastAPI, UploadFile

app = FastAPI()
qa_system = DocumentQASystem()

@app.post("/api/documents/upload")
async def upload_document(file: UploadFile):
    file_path = f"/tmp/{file.filename}"
    with open(file_path, "wb") as f:
        f.write(await file.read())
    
    file_type = file.filename.split('.')[-1]
    return await qa_system.ingest_document(file_path, file_type)

@app.post("/api/query")
async def query(question: str):
    return await qa_system.query(question)
```
