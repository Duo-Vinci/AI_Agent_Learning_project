# 智能客服系统案例

## 一、项目背景

某电商平台每天接收3万+客户咨询，人工客服压力巨大。决定构建AI智能客服系统，要求：
- 7×24小时在线服务
- 准确率>90%
- 平均响应时间<3秒
- 支持多轮对话
- 能够查询订单、退换货等

---

## 二、技术架构

### 2.1 整体架构图

```
用户查询
    ↓
[意图识别] → [简单FAQ] → 直接回答
    ↓
[知识库检索(RAG)]
    ↓
[Agent决策]
    ├─ 订单查询工具
    ├─ 退款处理工具
    ├─ 物流查询工具
    └─ 人工转接
    ↓
[对话管理] → 保存历史
    ↓
返回用户
```

### 2.2 技术选型

- **LLM**: DeepSeek-v4 (性价比高)
- **RAG**: FAISS + BGE Embeddings
- **框架**: LangChain + FastAPI
- **数据库**: PostgreSQL + Redis
- **部署**: Docker + K8s

## 三、核心功能实现

### 3.1 意图识别

```python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate

class IntentClassifier:
    """意图识别器"""
    
    def __init__(self):
        self.llm = ChatOpenAI(
            model="deepseek-chat",
            temperature=0
        )
        
        self.intents = {
            'order_query': '订单查询',
            'refund': '退款退货',
            'logistics': '物流查询',
            'product_info': '商品咨询',
            'complaint': '投诉建议',
            'other': '其他'
        }
    
    async def classify(self, query: str, history: List[dict] = None) -> str:
        """分类用户意图"""
        
        context = ""
        if history:
            context = f"\n对话历史：\n" + "\n".join([
                f"用户: {m['user']}\n客服: {m['assistant']}"
                for m in history[-3:]  # 最近3轮
            ])
        
        prompt = f"""你是一个意图识别专家。根据用户查询判断其意图。

可选意图：
{json.dumps(self.intents, ensure_ascii=False, indent=2)}

{context}

用户查询：{query}

直接返回意图类型（如：order_query），不要解释。
意图类型："""
        
        result = await self.llm.agenerate(prompt)
        intent = result.strip()
        
        return intent if intent in self.intents else 'other'
```

### 3.2 知识库RAG系统

```python
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter

class KnowledgeBase:
    """知识库"""
    
    def __init__(self):
        # 使用本地Embedding模型（节省成本）
        self.embeddings = HuggingFaceEmbeddings(
            model_name="BAAI/bge-large-zh-v1.5",
            model_kwargs={'device': 'cpu'}
        )
        
        self.vectorstore = None
        self.llm = ChatOpenAI(model="deepseek-chat", temperature=0.3)
    
    def build_from_documents(self, documents: List[str]):
        """构建知识库"""
        # 文档分块
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=500,
            chunk_overlap=50,
            separators=["\n\n", "\n", "。", ".", " "]
        )
        
        chunks = []
        for doc in documents:
            chunks.extend(splitter.split_text(doc))
        
        # 创建向量库
        self.vectorstore = FAISS.from_texts(
            chunks,
            self.embeddings
        )
        
        # 保存到磁盘
        self.vectorstore.save_local("kb_index")
    
    def load(self):
        """加载知识库"""
        self.vectorstore = FAISS.load_local(
            "kb_index",
            self.embeddings
        )
    
    async def query(self, question: str, k: int = 3) -> str:
        """查询知识库"""
        # 检索相关文档
        docs = self.vectorstore.similarity_search(question, k=k)
        
        # 构建上下文
        context = "\n\n".join([doc.page_content for doc in docs])
        
        # 生成答案
        prompt = f"""根据以下知识库内容回答用户问题。

知识库内容：
{context}

用户问题：{question}

要求：
1. 答案必须基于知识库内容
2. 如果知识库中没有相关信息，说"抱歉，这个问题我需要为您转接人工客服"
3. 回答简洁专业

回答："""
        
        answer = await self.llm.agenerate(prompt)
        return answer
```

### 3.3 工具系统

```python
from langchain.tools import tool
import requests

@tool
def query_order(order_id: str) -> dict:
    """查询订单状态
    
    Args:
        order_id: 订单号
    
    Returns:
        订单信息字典
    """
    # 调用订单系统API
    response = requests.get(
        f"https://api.example.com/orders/{order_id}",
        headers={"Authorization": "Bearer xxx"}
    )
    
    if response.status_code == 200:
        order = response.json()
        return {
            "status": "success",
            "order_status": order['status'],
            "payment_status": order['payment_status'],
            "logistics": order.get('logistics_info', 'N/A')
        }
    else:
        return {"status": "error", "message": "订单不存在"}

@tool
def query_logistics(tracking_number: str) -> dict:
    """查询物流信息"""
    response = requests.get(
        f"https://api.logistics.com/track/{tracking_number}"
    )
    
    if response.status_code == 200:
        data = response.json()
        return {
            "status": "success",
            "current_location": data['location'],
            "estimated_delivery": data['estimated_time'],
            "trace": data['trace'][-3:]  # 最近3条
        }
    else:
        return {"status": "error", "message": "物流信息查询失败"}

@tool
def create_refund(order_id: str, reason: str) -> dict:
    """创建退款申请"""
    response = requests.post(
        "https://api.example.com/refunds",
        json={
            "order_id": order_id,
            "reason": reason
        }
    )
    
    if response.status_code == 200:
        return {
            "status": "success",
            "refund_id": response.json()['refund_id'],
            "message": "退款申请已提交，预计3-5个工作日处理"
        }
    else:
        return {"status": "error", "message": "退款申请失败"}
```

### 3.4 Agent主控

```python
from langchain.agents import AgentExecutor, create_react_agent
from langchain import hub

class CustomerServiceAgent:
    """客服Agent"""
    
    def __init__(self):
        self.llm = ChatOpenAI(model="deepseek-chat", temperature=0.7)
        self.kb = KnowledgeBase()
        self.kb.load()
        
        # 工具列表
        self.tools = [
            query_order,
            query_logistics,
            create_refund,
        ]
        
        # 创建Agent
        prompt = hub.pull("hwchase17/react")
        agent = create_react_agent(self.llm, self.tools, prompt)
        
        self.agent_executor = AgentExecutor(
            agent=agent,
            tools=self.tools,
            verbose=True,
            max_iterations=5,
            handle_parsing_errors=True
        )
    
    async def handle_query(
        self,
        user_id: str,
        query: str,
        session_id: str
    ) -> dict:
        """处理用户查询"""
        
        # 1. 加载对话历史
        history = await self._load_history(session_id)
        
        # 2. 意图识别
        classifier = IntentClassifier()
        intent = await classifier.classify(query, history)
        
        # 3. 根据意图路由
        if intent in ['product_info', 'other']:
            # 简单问题：直接用知识库
            answer = await self.kb.query(query)
            tool_used = None
        
        else:
            # 复杂问题：使用Agent + 工具
            result = await self.agent_executor.ainvoke({
                "input": query,
                "chat_history": self._format_history(history)
            })
            
            answer = result['output']
            tool_used = self._extract_tools_used(result)
        
        # 4. 保存历史
        await self._save_history(session_id, query, answer)
        
        return {
            "answer": answer,
            "intent": intent,
            "tools_used": tool_used,
            "session_id": session_id
        }
```

### 3.5 对话管理

```python
import redis
import json

class ConversationManager:
    """对话管理器"""
    
    def __init__(self):
        self.redis = redis.Redis(
            host='localhost',
            port=6379,
            decode_responses=True
        )
        self.ttl = 3600  # 对话保留1小时
    
    async def save_turn(
        self,
        session_id: str,
        user_message: str,
        assistant_message: str,
        metadata: dict = None
    ):
        """保存一轮对话"""
        key = f"session:{session_id}"
        
        turn = {
            "user": user_message,
            "assistant": assistant_message,
            "timestamp": datetime.now().isoformat(),
            "metadata": metadata or {}
        }
        
        # 添加到列表
        self.redis.rpush(key, json.dumps(turn, ensure_ascii=False))
        
        # 设置过期时间
        self.redis.expire(key, self.ttl)
    
    async def get_history(
        self,
        session_id: str,
        limit: int = 10
    ) -> List[dict]:
        """获取对话历史"""
        key = f"session:{session_id}"
        
        # 获取最近N条
        history = self.redis.lrange(key, -limit, -1)
        
        return [json.loads(turn) for turn in history]
    
    async def clear_session(self, session_id: str):
        """清除会话"""
        key = f"session:{session_id}"
        self.redis.delete(key)
```

## 四、API服务

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI(title="智能客服API")

class QueryRequest(BaseModel):
    user_id: str
    query: str
    session_id: str = None

class QueryResponse(BaseModel):
    answer: str
    intent: str
    session_id: str
    timestamp: str

# 全局Agent实例
agent = CustomerServiceAgent()
conversation_mgr = ConversationManager()

@app.post("/api/v1/chat", response_model=QueryResponse)
async def chat(request: QueryRequest):
    """聊天接口"""
    
    # 生成session_id
    if not request.session_id:
        request.session_id = f"{request.user_id}_{int(time.time())}"
    
    try:
        # 处理查询
        result = await agent.handle_query(
            request.user_id,
            request.query,
            request.session_id
        )
        
        # 保存对话
        await conversation_mgr.save_turn(
            request.session_id,
            request.query,
            result['answer'],
            {"intent": result['intent']}
        )
        
        return QueryResponse(
            answer=result['answer'],
            intent=result['intent'],
            session_id=request.session_id,
            timestamp=datetime.now().isoformat()
        )
    
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/api/v1/history/{session_id}")
async def get_history(session_id: str):
    """获取对话历史"""
    history = await conversation_mgr.get_history(session_id)
    return {"history": history}
```

## 七、经验总结

### 7.1 成功关键

1. **准确的意图识别**：快速路由到正确处理流程
2. **完善的知识库**：80%问题靠知识库解决
3. **工具集成**：与业务系统深度集成
4. **对话管理**：支持多轮对话和上下文
5. **降级机制**：AI无法处理时及时转人工

### 7.2 遇到的挑战

**挑战1：知识库更新滞后**
- 解决：每周自动从工单系统更新知识库

**挑战2：复杂订单问题处理不当**
- 解决：细化意图分类，增加专门的订单Agent

**挑战3：高峰期响应慢**
- 解决：添加缓存、增加实例、负载均衡

### 7.3 持续优化

- 收集用户反馈，标注错误case
- 定期重新训练意图识别
- A/B测试不同的Prompt
- 监控成本和性能指标
