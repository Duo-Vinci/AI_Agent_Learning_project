# 成本控制策略

## 1. 概述

AI Agent应用的成本主要来源于API调用、Token使用和计算资源消耗。本文档提供全面的成本控制策略，帮助开发者在保证服务质量的同时，有效降低运营成本。

### 1.1 成本构成分析

**典型AI应用的成本结构**：
- API调用费用：60-70%
- Token使用费用：20-30%
- 基础设施费用：5-10%
- 存储和带宽费用：3-5%

**不同模型的成本对比**（2024年价格）：

| 模型 | 输入价格(/1M tokens) | 输出价格(/1M tokens) | 适用场景 |
|------|---------------------|---------------------|---------|
| GPT-4 Turbo | $10.00 | $30.00 | 复杂推理、高质量输出 |
| GPT-3.5 Turbo | $0.50 | $1.50 | 一般对话、简单任务 |
| Claude 3 Opus | $15.00 | $75.00 | 最高质量要求 |
| Claude 3 Sonnet | $3.00 | $15.00 | 平衡性能和成本 |
| Claude 3 Haiku | $0.25 | $1.25 | 高频简单任务 |

```python
class SlidingWindowContext:
    """滑动窗口上下文管理器"""
    
    def __init__(self, max_tokens=2000):
        self.max_tokens = max_tokens
        self.messages = []
    
    def add_message(self, role, content):
        """添加消息"""
        self.messages.append({"role": role, "content": content})
        self._trim_context()
    
    def _estimate_tokens(self, text):
        """估算token数（简化版）"""
        return len(text) // 4  # 粗略估算：4字符≈1token
    
    def _trim_context(self):
        """修剪上下文，保持在限制内"""
        total_tokens = sum(self._estimate_tokens(msg["content"]) 
                          for msg in self.messages)
        
        # 如果超出限制，删除最旧的消息（保留system message）
        while total_tokens > self.max_tokens and len(self.messages) > 1:
            if self.messages[0]["role"] == "system":
                # 保留system message，删除第二条
                if len(self.messages) > 1:
                    removed = self.messages.pop(1)
            else:
                removed = self.messages.pop(0)
            
            total_tokens -= self._estimate_tokens(removed["content"])
    
    def get_messages(self):
        """获取当前消息列表"""
        return self.messages
    
    def get_token_count(self):
        """获取当前token估算"""
        return sum(self._estimate_tokens(msg["content"]) 
                  for msg in self.messages)

# 使用示例

context.add_message("system", "你是AI助手")
```

```python
class ContextCompressor:
    """上下文压缩器"""
    
    def __init__(self, client):
        self.client = client
    
    def compress_history(self, messages, target_ratio=0.3):
        """压缩对话历史"""
        # 提取需要压缩的内容
        history_text = "
".join(
            f"{msg['role']}: {msg['content']}" 
            for msg in messages[1:-1]  # 保留system和最后一条
        )
        
        # 使用便宜的模型生成摘要
        summary_prompt = f"""请将以下对话历史压缩为简短摘要，保留关键信息：

{history_text}

压缩摘要："""
        
        response = self.client.chat.completions.create(
            model="gpt-3.5-turbo",
            messages=[{"role": "user", "content": summary_prompt}],
            max_tokens=int(len(history_text) * target_ratio / 4),
            temperature=0.3
        )
        
        summary = response.choices[0].message.content
        
        # 构建压缩后的消息列表
        compressed_messages = [
            messages[0],  # system message
            {"role": "system", "content": f"对话历史摘要：{summary}"},
            messages[-1]  # 最新消息
        ]
        
        return compressed_messages
    
    def should_compress(self, messages, threshold=3000):
        """判断是否需要压缩"""
        total_length = sum(len(msg["content"]) for msg in messages)
        return total_length > threshold
```

### 3.1 批量处理

**策略1：请求批处理**

```python
class BatchProcessor:
    """批量处理器"""
    
    def __init__(self, client, batch_size=10):
        self.client = client
        self.batch_size = batch_size
        self.queue = []
    
    def add_request(self, prompt, callback=None):
        """添加请求到队列"""
        self.queue.append({"prompt": prompt, "callback": callback})
        
        # 达到批次大小时自动处理
        if len(self.queue) >= self.batch_size:
            return self.process_batch()
        
        return None
    
    def process_batch(self):
        """批量处理请求"""
        if not self.queue:
            return []
        
        # 构建批量Prompt
        batch_prompt = "请分别回答以下问题，用编号标识：

"
        for i, item in enumerate(self.queue, 1):
            batch_prompt += f"{i}. {item['prompt']}
"
        
        # 单次API调用处理所有请求
        response = self.client.chat.completions.create(
            model="gpt-3.5-turbo",
            messages=[{"role": "user", "content": batch_prompt}],
            max_tokens=200 * len(self.queue)
        )
        
        # 解析响应
        result_text = response.choices[0].message.content
        results = self._parse_batch_response(result_text, len(self.queue))
        
        # 执行回调
        for item, result in zip(self.queue, results):
            if item["callback"]:
                item["callback"](result)
        
        self.queue.clear()
        return results
    
    def _parse_batch_response(self, text, count):
        """解析批量响应"""
        results = []
        lines = text.split("
")
        current_result = []
        
        for line in lines:
            if line.strip() and line[0].isdigit() and ". " in line[:5]:
                if current_result:
                    results.append("
".join(current_result))
                    current_result = []
                current_result.append(line.split(". ", 1)[1])
            elif current_result:
                current_result.append(line)
        
        if current_result:
            results.append("
".join(current_result))
        
        return results[:count]

# 使用示例
processor = BatchProcessor(client, batch_size=5)

# 添加多个请求
questions = [
    "什么是Python？",
    "什么是JavaScript？",
    "什么是Go？",
    "什么是Rust？",
    "什么是TypeScript？"
]

for q in questions:
    processor.add_request(q, callback=lambda r: print(f"结果: {r}"))

# 成本分析：
# 单独调用：5次 × $0.0003 = $0.0015
# 批量调用：1次 × $0.0008 = $0.0008
# 节省：47%
```

**策略2：异步批量处理**

```python
import asyncio
from openai import AsyncOpenAI

class AsyncBatchProcessor:
    """异步批量处理器"""
    
    def __init__(self, api_key, max_concurrent=5):
        self.client = AsyncOpenAI(api_key=api_key)
        self.max_concurrent = max_concurrent
        self.semaphore = asyncio.Semaphore(max_concurrent)
    
    async def process_single(self, prompt):
        """处理单个请求"""
        async with self.semaphore:
            response = await self.client.chat.completions.create(
                model="gpt-3.5-turbo",
                messages=[{"role": "user", "content": prompt}],
                max_tokens=150
            )
            return response.choices[0].message.content
    
    async def process_many(self, prompts):
        """并发处理多个请求"""
        tasks = [self.process_single(prompt) for prompt in prompts]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        return results

# 使用示例

    prompts = [f"解释概念{i}" for i in range(20)]

    results = await processor.process_many(prompts)
    duration = time.time() - start
    
    print(f"处理{len(prompts)}个请求耗时: {duration:.2f}秒")
    print(f"平均每个请求: {duration/len(prompts):.2f}秒")
    
    # 性能提升：
    # 串行：20 × 2秒 = 40秒
```

### 3.2 请求去重

**策略：智能缓存相似请求**

```python
import hashlib
from datetime import datetime, timedelta

class RequestDeduplicator:
    def __init__(self, client, cache_ttl=3600):
        self.client = client
        self.cache = {}
        self.cache_ttl = cache_ttl
    
    def process_request(self, prompt):
        cache_key = hashlib.md5(prompt.encode()).hexdigest()
        
        # 检查缓存
        if cache_key in self.cache:
            cache_data = self.cache[cache_key]
            if datetime.now() <= cache_data["expires"]:
                return cache_data["response"]
        
        # API调用
        response = self.client.chat.completions.create(
            model="gpt-3.5-turbo",
            messages=[{"role": "user", "content": prompt}]
        )
        
        result = response.choices[0].message.content
        
        # 缓存结果
        self.cache[cache_key] = {
            "response": result,
            "expires": datetime.now() + timedelta(seconds=self.cache_ttl)
        }
        
        return result

# 成本节省：重复请求场景下节省50-70%
```

## 4. 缓存策略降低成本

### 4.1 多层缓存架构

**L1（内存）+ L2（Redis）+ L3（数据库）**

```python
import redis
import json
import hashlib

class MultiLevelCache:
    def __init__(self, redis_client=None):
        self.redis_client = redis_client
        self.l1_cache = {}
        self.stats = {"l1_hits": 0, "l2_hits": 0, "misses": 0}
    
    def _get_key(self, prompt, model):
        return hashlib.sha256(f"{prompt}:{model}".encode()).hexdigest()
    
    def get(self, prompt, model):
        key = self._get_key(prompt, model)
        
        # L1: 内存缓存（最快）
        if key in self.l1_cache:
            self.stats["l1_hits"] += 1
            return self.l1_cache[key]
        
        # L2: Redis缓存
        if self.redis_client:
            data = self.redis_client.get(key)
            if data:
                result = json.loads(data)
                self.l1_cache[key] = result
                self.stats["l2_hits"] += 1
                return result
        
        self.stats["misses"] += 1
        return None
    
    def set(self, prompt, model, value, ttl=3600):
        key = self._get_key(prompt, model)
        
        # 写入L1
        self.l1_cache[key] = value
        
        # 写入L2
        if self.redis_client:
            self.redis_client.setex(key, ttl, json.dumps(value))

# 使用示例
redis_client = redis.Redis(host="localhost", port=6379)
cache = MultiLevelCache(redis_client)

def cached_query(prompt):
    # 检查缓存
    cached = cache.get(prompt, "gpt-3.5-turbo")
    if cached:
        return cached
    
    # API调用
    response = client.chat.completions.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}]
    )
    
    result = response.choices[0].message.content
    cache.set(prompt, "gpt-3.5-turbo", result)
    
    return result

# 性能提升：
# - L1命中：~0.1ms，成本$0
# - L2命中：~1ms，成本$0
# - API调用：~500ms，成本$0.0003
# 缓存命中率50%时节省50%成本
```

### 4.2 语义缓存

```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

class SemanticCache:
    def __init__(self, client, similarity_threshold=0.92):
        self.client = client
        self.cache = []
        self.threshold = similarity_threshold
    
    def get(self, prompt):
        if not self.cache:
            return None
        
        prompt_emb = self._get_embedding(prompt)
        
        for item in self.cache:
            similarity = cosine_similarity(
                prompt_emb.reshape(1, -1),
                item["embedding"].reshape(1, -1)
            )[0][0]
            
            if similarity >= self.threshold:
                return item["response"]
        
        return None

# 成本节省：语义相似请求命中率60-70%
```

```python
        self.current_cost = 0
```

```python
class QuotaManager:
    def __init__(self, daily_budget=100.0):
        self.daily_budget = daily_budget
        self.user_quotas = {}
        self.total_used = 0
    
    def set_user_quota(self, user_id, quota):
        self.user_quotas[user_id] = {
            "limit": quota,
            "used": 0
        }
    
    def check_quota(self, user_id, estimated_cost):
        # 检查用户配额
        if user_id in self.user_quotas:
            user_quota = self.user_quotas[user_id]
            if user_quota["used"] + estimated_cost > user_quota["limit"]:
                return False, "用户配额不足"
        
        # 检查总预算
        if self.total_used + estimated_cost > self.daily_budget:
            return False, "系统预算不足"
        
        return True, "OK"
    
    def consume_quota(self, user_id, actual_cost):
        if user_id in self.user_quotas:
            self.user_quotas[user_id]["used"] += actual_cost
        
        self.total_used += actual_cost
    
    def get_user_stats(self, user_id):
        if user_id not in self.user_quotas:
            return None
        
        quota = self.user_quotas[user_id]
        return {
            "limit": quota["limit"],
            "used": quota["used"],
            "remaining": quota["limit"] - quota["used"],
            "usage_rate": f"{quota['used']/quota['limit']*100:.1f}%"
        }

# 使用示例
quota_mgr = QuotaManager(daily_budget=1000.0)

# 设置用户配额

quota_mgr.set_user_quota("user_002", quota=50.0)
```

```python
from datetime import datetime
import json

class CostReportGenerator:
    def __init__(self):
        self.records = []
    
    def add_record(self, user_id, model, cost, timestamp=None):
        if timestamp is None:
            timestamp = datetime.now()
        
        self.records.append({
            "user_id": user_id,
            "model": model,
            "cost": cost,
            "timestamp": timestamp
        })
    
    def generate_daily_report(self, date=None):
        if date is None:
            date = datetime.now().date()
        
        daily_records = [
            r for r in self.records
            if r["timestamp"].date() == date
        ]
        
        if not daily_records:
            return {"error": "No data for this date"}
        
        total_cost = sum(r["cost"] for r in daily_records)
        total_requests = len(daily_records)
        
        # 按模型分组
        by_model = {}
        for r in daily_records:
            model = r["model"]
            if model not in by_model:
                by_model[model] = {"count": 0, "cost": 0}
            by_model[model]["count"] += 1
            by_model[model]["cost"] += r["cost"]
        
        # 按用户分组
        by_user = {}
        for r in daily_records:
            user = r["user_id"]
            if user not in by_user:
                by_user[user] = {"count": 0, "cost": 0}
            by_user[user]["count"] += 1
            by_user[user]["cost"] += r["cost"]
        
        # Top用户
        top_users = sorted(
            by_user.items(),
            key=lambda x: x[1]["cost"],
            reverse=True
        )[:5]
        
        return {
            "date": str(date),
            "total_cost": f"${total_cost:.2f}",
            "total_requests": total_requests,
            "avg_cost_per_request": f"${total_cost/total_requests:.6f}",
            "by_model": by_model,
            "top_users": [
                {"user_id": u[0], "cost": f"${u[1]['cost']:.2f}", "requests": u[1]["count"]}
                for u in top_users
            ]
        }
    
    def export_report(self, filename):
        report = self.generate_daily_report()
        
        with open(filename, 'w', encoding='utf-8') as f:

        return filename
```

```python
from datetime import datetime, timedelta

class CostTrendAnalyzer:
    def __init__(self):
        self.daily_costs = {}
    
    def add_daily_cost(self, date, cost):
        date_str = date.strftime("%Y-%m-%d")
        if date_str not in self.daily_costs:
            self.daily_costs[date_str] = 0
        self.daily_costs[date_str] += cost
    
    def get_trend(self, days=7):
        end_date = datetime.now()
        dates = []
        costs = []
        
        for i in range(days):
            date = end_date - timedelta(days=i)
            date_str = date.strftime("%Y-%m-%d")
            dates.append(date_str)
            costs.append(self.daily_costs.get(date_str, 0))
        
        dates.reverse()
        costs.reverse()
        
        return {"dates": dates, "costs": costs}
    
    def predict_monthly_cost(self):
        trend = self.get_trend(days=7)
        avg_daily = sum(trend["costs"]) / len(trend["costs"])
        return avg_daily * 30

# 使用示例

for i in range(7):

    cost = 10 + i * 2
    analyzer.add_daily_cost(date, cost)
```

```python
class CodeAssistantOptimizer:
    def __init__(self, client):
        self.client = client
        self.code_cache = {}
        self.common_patterns = {
            "hello_world": "print('Hello, World!')",
            "read_file": "with open('file.txt', 'r') as f:\n    content = f.read()",
        }
    
    def get_code_completion(self, prompt, max_tokens=500):
        # 检查常见模式
        for pattern, code in self.common_patterns.items():
            if pattern.replace("_", " ") in prompt.lower():
                return {"source": "pattern", "code": code, "cost": 0}
        
        # 检查缓存
        if prompt in self.code_cache:
            return {"source": "cache", "code": self.code_cache[prompt], "cost": 0}
```
