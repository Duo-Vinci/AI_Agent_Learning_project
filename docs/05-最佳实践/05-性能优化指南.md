## 二、LLM 调用优化

### 2.1 减少 Token 使用

```python
class PromptOptimizer:
    """Prompt 优化器"""
    
    def optimize(self, prompt: str) -> str:
        """优化 Prompt 减少 Token"""
        # 1. 移除多余空白
        prompt = ' '.join(prompt.split())
        
        # 2. 使用缩写（保持清晰度）
        abbreviations = {
            'for example': 'e.g.',
            'that is': 'i.e.',
            'and so on': 'etc.',
        }
        for full, abbr in abbreviations.items():
            prompt = prompt.replace(full, abbr)
        
        # 3. 移除冗余词汇
        redundant_phrases = [
            'please', 'kindly', 'I would like to',
            'could you', 'would you mind'
        ]
        for phrase in redundant_phrases:
            prompt = prompt.replace(phrase, '')
        
        return prompt.strip()

# 效果对比
original = "Please could you kindly analyze the following text for me"
optimized = "Analyze the following text"
# Token 减少: 12 → 5 (节省58%)
```

## 三、缓存策略

### 3.1 多层缓存

```python
class MultiLevelCache:
    """多层缓存系统"""
    
    def __init__(self):
        self.memory_cache = {}  # L1: 内存缓存
        self.redis_client = redis.Redis()  # L2: Redis
        self.max_memory_size = 1000
    
    async def get(self, key: str):
        """获取缓存（优先内存）"""
        # L1: 内存
        if key in self.memory_cache:
            return self.memory_cache[key]
        
        # L2: Redis
        value = await self.redis_client.get(key)
        if value:
            # 回填到内存
            if len(self.memory_cache) < self.max_memory_size:
                self.memory_cache[key] = value
            return value
        
        return None
    
    async def set(self, key: str, value: Any, ttl: int = 3600):
        """设置缓存"""
        # 写入 L1
        if len(self.memory_cache) < self.max_memory_size:
            self.memory_cache[key] = value
        
        # 写入 L2
        await self.redis_client.setex(key, ttl, value)

# 性能提升
# L1 命中: <1ms
# L2 命中: ~5ms  
# Cache miss: >1000ms (LLM调用)
```

### 3.2 智能缓存键

```python
import hashlib
import json

def make_cache_key(
    model: str,
    prompt: str,
    temperature: float,
    **kwargs
) -> str:
    """生成标准化的缓存键"""
    # 归一化参数
    cache_data = {
        'model': model,
        'prompt': prompt.strip(),
        'temperature': round(temperature, 2),
        'params': {k: v for k, v in sorted(kwargs.items())}
    }
    
    # 生成哈希
    data_str = json.dumps(cache_data, sort_keys=True)
    return hashlib.md5(data_str.encode()).hexdigest()
```

### 4.2 并发控制

```python
import asyncio

class RateLimiter:
    """限流器"""
    
    def __init__(self, max_concurrent: int = 10, rate_per_minute: int = 60):
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.rate_limit = rate_per_minute
        self.tokens = rate_per_minute
        self.last_update = time.time()
    
    async def acquire(self):
        """获取许可"""
        async with self.semaphore:
            # 令牌桶算法
            now = time.time()
            elapsed = now - self.last_update
            self.tokens = min(
                self.rate_limit,
                self.tokens + elapsed * (self.rate_limit / 60)
            )
            self.last_update = now
            
            if self.tokens < 1:
                sleep_time = (1 - self.tokens) * (60 / self.rate_limit)
                await asyncio.sleep(sleep_time)
                self.tokens = 0
            else:
                self.tokens -= 1

# 使用
limiter = RateLimiter(max_concurrent=10, rate_per_minute=60)

async def limited_call(query: str):
    await limiter.acquire()
    return await llm.agenerate(query)
```

## 六、向量检索优化

### 6.1 FAISS 索引选择

```python
def create_optimized_faiss_index(
    embeddings: np.ndarray,
    index_type: str = 'auto'
):
    """创建优化的 FAISS 索引"""
    dim = embeddings.shape[1]
    n = embeddings.shape[0]
    
    if index_type == 'auto':
        if n < 10000:
            # Flat: 精确但慢
            index = faiss.IndexFlatL2(dim)
        elif n < 1000000:
            # IVF: 平衡
            nlist = int(np.sqrt(n))
            quantizer = faiss.IndexFlatL2(dim)
            index = faiss.IndexIVFFlat(quantizer, dim, nlist)
            index.nprobe = 10  # 查询时探测的聚类数
        else:
            # HNSW: 快速近似
            index = faiss.IndexHNSWFlat(dim, 32)
            index.hnsw.efSearch = 64
    
    index.train(embeddings)
    index.add(embeddings)
    
    return index

# 性能对比（100万向量）
# Flat: 100% 精度, 1000ms
# IVF:  98% 精度, 50ms (20x faster)
# HNSW: 95% 精度, 5ms (200x faster)
```

```python
from queue import Queue

class ObjectPool:
    """对象池"""
    
    def __init__(self, factory, size: int = 10):
        self.factory = factory
        self.pool = Queue(maxsize=size)
        
        # 预创建对象
        for _ in range(size):
            self.pool.put(factory())
    
    def acquire(self):
        """获取对象"""
        return self.pool.get()
    
    def release(self, obj):
        """释放对象"""
        # 重置对象状态
        obj.reset()
        self.pool.put(obj)
```

```python
import time
from contextlib import contextmanager

@contextmanager
def timer(name: str):
    """计时上下文管理器"""
    start = time.perf_counter()
    try:
        yield
    finally:
        duration = time.perf_counter() - start
        print(f"{name}: {duration*1000:.2f}ms")

# 使用
```
