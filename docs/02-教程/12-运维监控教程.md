```python
# src/utils/metrics.py
from prometheus_client import Counter, Histogram, Gauge, Info
from prometheus_client import start_http_server
import time

# 定义指标
agent_requests_total = Counter(
    'agent_requests_total',
    'Total number of agent requests',
    ['agent_id', 'status']
)

agent_request_duration_seconds = Histogram(
    'agent_request_duration_seconds',
    'Agent request duration in seconds',
    ['agent_id'],
    buckets=[0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 30.0, 60.0]
)

llm_tokens_total = Counter(
    'llm_tokens_total',
    'Total LLM tokens used',
    ['model', 'type']  # type: prompt or completion
)

llm_cost_total = Counter(
    'llm_cost_total',
    'Total LLM cost in USD',
    ['model']
)

active_agents = Gauge(
    'active_agents',
    'Number of currently active agents'
)

agent_info = Info(
    'agent_info',
    'Agent information'
)

class MetricsCollector:
    """指标收集器"""
    
    def __init__(self, port: int = 8001):
        self.port = port
        start_http_server(port)
    
    def record_request(self, agent_id: str, status: str, duration: float):
        """记录请求"""
        agent_requests_total.labels(agent_id=agent_id, status=status).inc()
        agent_request_duration_seconds.labels(agent_id=agent_id).observe(duration)
    
    def record_llm_usage(self, model: str, prompt_tokens: int, completion_tokens: int):
        """记录 LLM 使用"""
        llm_tokens_total.labels(model=model, type='prompt').inc(prompt_tokens)
        llm_tokens_total.labels(model=model, type='completion').inc(completion_tokens)
        
        # 计算成本
        cost = self._calculate_cost(model, prompt_tokens, completion_tokens)
        llm_cost_total.labels(model=model).inc(cost)
    
    def _calculate_cost(self, model: str, prompt_tokens: int, completion_tokens: int) -> float:
        """计算成本"""
        pricing = {
            'gpt-4': {'prompt': 0.03/1000, 'completion': 0.06/1000},
            'gpt-3.5-turbo': {'prompt': 0.0015/1000, 'completion': 0.002/1000},
            'deepseek-chat': {'prompt': 0.0001/1000, 'completion': 0.0002/1000}
        }
        rates = pricing.get(model, pricing['gpt-4'])
        return prompt_tokens * rates['prompt'] + completion_tokens * rates['completion']

# 使用示例
collector = MetricsCollector(port=8001)

async def execute_with_metrics(agent_id: str, task: dict):
    """带指标的执行"""
    start_time = time.time()
    active_agents.inc()
    
    try:
        result = await agent.execute(task)
        duration = time.time() - start_time
        
        collector.record_request(agent_id, 'success', duration)
        
        # 记录 LLM 使用
        if 'usage' in result:
            collector.record_llm_usage(
                model=result['model'],
                prompt_tokens=result['usage']['prompt_tokens'],
                completion_tokens=result['usage']['completion_tokens']
            )
        
        return result
    
    except Exception as e:
        duration = time.time() - start_time
        collector.record_request(agent_id, 'error', duration)
        raise
    
    finally:
        active_agents.dec()
```

### 4.1 OpenTelemetry 集成

```python
# src/utils/tracing.py
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.jaeger.thrift import JaegerExporter
from opentelemetry.sdk.resources import Resource

def setup_tracing(service_name: str = "ai-agent"):
    """配置分布式追踪"""
    
    # 设置资源信息
    resource = Resource.create({
        "service.name": service_name,
        "service.version": "1.0.0"
    })
    
    # 创建 TracerProvider
    provider = TracerProvider(resource=resource)
    
    # 配置 Jaeger exporter
    jaeger_exporter = JaegerExporter(
        agent_host_name="localhost",
        agent_port=6831,
    )
    
    # 添加 span processor
    provider.add_span_processor(
        BatchSpanProcessor(jaeger_exporter)
    )
    
    # 设置全局 tracer provider
    trace.set_tracer_provider(provider)
    
    return trace.get_tracer(__name__)

# 使用示例
tracer = setup_tracing()

async def execute_agent_with_tracing(agent_id: str, task: dict):
    """带追踪的 Agent 执行"""
    
    with tracer.start_as_current_span("agent.execute") as span:
        span.set_attribute("agent.id", agent_id)
        span.set_attribute("task.type", task.get("type"))
        
        try:
            # LLM 调用
            with tracer.start_as_current_span("llm.call") as llm_span:
                llm_span.set_attribute("llm.model", "gpt-4")
                result = await call_llm(task["prompt"])
                llm_span.set_attribute("llm.tokens", result["usage"]["total_tokens"])
            
            # 工具调用
            if task.get("use_tools"):
                with tracer.start_as_current_span("tools.execute") as tool_span:
                    tool_span.set_attribute("tool.name", task["tool"])
                    tool_result = await execute_tool(task["tool"])
            
            span.set_attribute("status", "success")
            return result
            
        except Exception as e:
            span.set_attribute("error", True)
            span.set_attribute("error.message", str(e))
            span.record_exception(e)
            raise
```

```python
import cProfile
import pstats
import io
from functools import wraps

def profile(func):
    """性能分析装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        profiler = cProfile.Profile()
        profiler.enable()
        
        try:
            result = func(*args, **kwargs)
            return result
        finally:
            profiler.disable()
            
            # 输出结果
            s = io.StringIO()
            ps = pstats.Stats(profiler, stream=s)
            ps.sort_stats('cumulative')
            ps.print_stats(20)  # 前20个最慢的函数
            
            print(s.getvalue())
    
    return wrapper
```

### 6.2 LLM 调用优化

```python
class LLMOptimizer:
    """LLM 调用优化器"""
    
    def __init__(self, cache_enabled: bool = True):
        self.cache_enabled = cache_enabled
        self.cache = {}
        self.stats = {
            'total_calls': 0,
            'cache_hits': 0,
            'total_tokens': 0,
            'total_cost': 0
        }
    
    def optimize_prompt(self, prompt: str) -> str:
        """优化 Prompt 减少 Token"""
        # 移除多余空白
        prompt = ' '.join(prompt.split())
        
        # 使用缩写
        replacements = {
            'please': 'pls',
            'information': 'info',
            'description': 'desc'
        }
        
        for old, new in replacements.items():
            prompt = prompt.replace(old, new)
        
        return prompt
    
    async def call_with_cache(self, model: str, prompt: str):
        """带缓存的调用"""
        self.stats['total_calls'] += 1
        
        if self.cache_enabled:
            cache_key = hash(f"{model}:{prompt}")
            
            if cache_key in self.cache:
                self.stats['cache_hits'] += 1
                return self.cache[cache_key]
        
        # 优化 Prompt
        optimized_prompt = self.optimize_prompt(prompt)
        
        # 调用 LLM
        result = await llm.agenerate(optimized_prompt)
        
        # 更新统计
        self.stats['total_tokens'] += result['usage']['total_tokens']
        self.stats['total_cost'] += self._calculate_cost(result)
        
        # 缓存结果
        if self.cache_enabled:
            self.cache[cache_key] = result
        
        return result
    
    def get_stats(self):
        """获取优化统计"""
        return {
            **self.stats,
            'cache_hit_rate': self.stats['cache_hits'] / max(self.stats['total_calls'], 1),
            'avg_tokens_per_call': self.stats['total_tokens'] / max(self.stats['total_calls'], 1)
        }
```

### 9.1 成本追踪

```python
class CostTracker:
    """成本追踪器"""
    
    def __init__(self):
        self.costs = {
            'llm': 0.0,
            'infrastructure': 0.0,
            'storage': 0.0
        }
        self.daily_limits = {
            'llm': 100.0,  # $100/day
            'total': 200.0  # $200/day
        }
    
    def track_llm_cost(self, model: str, tokens: dict):
        """追踪 LLM 成本"""
        cost = calculate_cost(model, tokens)
        self.costs['llm'] += cost
        
        # 检查是否超过预算
        if self.costs['llm'] > self.daily_limits['llm']:
            self._send_alert(f"LLM cost exceeded daily limit: ${self.costs['llm']:.2f}")
        
        return cost
    
    def get_report(self) -> dict:
        """生成成本报告"""
        total = sum(self.costs.values())
        
        return {
            'total_cost': round(total, 2),
            'breakdown': {k: round(v, 2) for k, v in self.costs.items()},
            'budget_used': round((total / self.daily_limits['total']) * 100, 2),
            'recommendations': self._get_recommendations()
        }
    
    def _get_recommendations(self) -> list:
        """成本优化建议"""
        recommendations = []
        
        if self.costs['llm'] > self.daily_limits['llm'] * 0.8:
            recommendations.append("考虑使用更便宜的模型（如 gpt-3.5-turbo）")
            recommendations.append("启用响应缓存减少重复调用")
            recommendations.append("优化 Prompt 减少 Token 使用")
        
        return recommendations
```
